50bool Graphics::Initialize(GLFWwindow* win)
51{
52 window = win;
53 glfwGetWindowSize(window, &width, &height);
54 projection = glm::perspective(glm::radians(fov), (float)width / (float)height, 0.1f, 100.0f);
55 SetViewport(0, 0, width, height);
56 glEnable(GL_DEPTH_TEST);
57
58 //Create Uniform Buffer Object for view and projection matrices
59 glGenBuffers(1, &uboMatrices);
60 glBindBuffer(GL_UNIFORM_BUFFER, uboMatrices);
61 glBufferData(GL_UNIFORM_BUFFER, 3 * sizeof(glm::mat4), NULL, GL_STATIC_DRAW);
62 glBindBuffer(GL_UNIFORM_BUFFER, 0);
63
64 //bind uniform buffer to binding point 0
65 glBindBufferRange(GL_UNIFORM_BUFFER, 0, uboMatrices, 0, 3 * sizeof(glm::mat4));
66
67 glBindBuffer(GL_UNIFORM_BUFFER, uboMatrices);
68 glBufferSubData(GL_UNIFORM_BUFFER, 0, sizeof(glm::mat4), glm::value_ptr(projection));
69 glBindBuffer(GL_UNIFORM_BUFFER, 0);
70
71 return true;
72}
