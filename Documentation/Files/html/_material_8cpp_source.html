86bool Material::SetShaders(unsigned int vertexShader, unsigned int fragmentShader)
87{
88 shaderProgram = glCreateProgram();
89 glAttachShader(shaderProgram, vertexShader);
90 glAttachShader(shaderProgram, fragmentShader);
91 glLinkProgram(shaderProgram);
92
93 // check for linking errors
94 int success;
95 char infoLog[512];
96 glGetProgramiv(shaderProgram, GL_LINK_STATUS, &success);
97 if (!success) {
98 glGetProgramInfoLog(shaderProgram, 512, NULL, infoLog);
99 std::cout << "ERROR::SHADER::PROGRAM::LINKING_FAILED\n" << infoLog << std::endl;
100 return false;
101 }
102
103 // Bind uniform blocks to binding points
104 GLuint ambientBlockIndex = glGetUniformBlockIndex(shaderProgram, "AmbientLightData");
105 if (ambientBlockIndex != GL_INVALID_INDEX) {
106 glUniformBlockBinding(shaderProgram, ambientBlockIndex, 1);
107 }
108
109 return true;
110}
