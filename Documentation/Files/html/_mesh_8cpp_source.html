19void Mesh::AddMeshData(std::vector<Vertex> &newVertices, std::vector<unsigned int> &newIndices)
20{
21
22 DestroyMeshData();
23
24 //Copy vertices
25 for (int i = 0; i < newVertices.size(); i++)
26 {
27 vertices.push_back(newVertices[i]);
28 }
29
30 //Copy indices
31 for (int i = 0; i < newIndices.size(); i++)
32 {
33 indices.push_back(newIndices[i]);
34 }
35
36 //create vao
37 glGenVertexArrays(1, &VAO);
38 glBindVertexArray(VAO);
39
40 //create vbo
41 glGenBuffers(1, &VBO);
42 glBindBuffer(GL_ARRAY_BUFFER, VBO);
43 glBufferData(GL_ARRAY_BUFFER, vertices.size() * sizeof(Vertex), &vertices[0], GL_STATIC_DRAW);
44
45 //create ebo
46 glGenBuffers(1, &EBO);
47 glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, EBO);
48 glBufferData(GL_ELEMENT_ARRAY_BUFFER, indices.size() * sizeof(unsigned int), &indices[0], GL_STATIC_DRAW);
49
50 //set vertex attribute layout
51 glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, sizeof(glm::vec3), (void*)0);
52 glEnableVertexAttribArray(0);
53
54}
